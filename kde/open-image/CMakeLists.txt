cmake_minimum_required(VERSION 3.24)
project(dwarfsctl_open LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
add_executable(dwarfsctl-open main.cpp)
target_include_directories(dwarfsctl-open PRIVATE ../common)
target_link_libraries(dwarfsctl-open PRIVATE Qt6::Widgets)
install(TARGETS dwarfsctl-open RUNTIME DESTINATION bin)
install(FILES dwarfsctl-open.desktop DESTINATION share/applications)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    set(CMAKE_AUTOMOC ON)
    add_executable(tst_open tests.cpp)
    target_include_directories(tst_open PRIVATE ../common)
    target_link_libraries(tst_open PRIVATE Qt6::Widgets Qt6::Test)
    add_test(NAME open_image COMMAND tst_open)
    set_tests_properties(open_image PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
